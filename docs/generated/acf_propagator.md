$$
q_\alpha = \sqrt{\frac{1}{v_p^2} - p^2},\qquad q_\beta = \sqrt{\frac{1}{v_s^2} - p^2}\qquad(\operatorname{Im} q \ge 0)
$$

$$
\text{P}: (u_x, u_z) = v_p\,(p,\ \pm q_\alpha),\qquad \text{S}: (u_x, u_z) = v_s\,(\pm q_\beta,\ -p)
$$

$$
\tau_{xz} = \mu\,(p\,u_z + q\,u_x),\qquad \tau_{zz} = \lambda\,(p\,u_x + q\,u_z) + 2\mu\, q\, u_z
$$

$$
\boxed{\;\mathbf v(z + h) = E\,\operatorname{diag}\!\left(e^{i\omega q_j h}\right) E^{-1}\,\mathbf v(z)\;}\qquad q_j = (q_\alpha, q_\beta, -q_\alpha, -q_\beta)
$$
